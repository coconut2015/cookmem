317 for (std::size_t i = 0; i < MEM_ALLOC_COUNT; ++i)
318 {
319 std::printf (" %s: %zu\n",
320 getAllocationTypeName (static_cast<MemAllocType_et> (i)),

374class StatsLogger
375{
376public:
378 void
379 logTrimEmptySegments ()
380 {
381 m_stats.setNumTrimEmptySegments (m_stats.getNumTrimEmptySegments () + 1);
382 }
383
385 void
386 logGetSegment (void* segment, std::size_t segmentSize)
387 {
388 if (segment == nullptr)
389 {
390 return;
391 }
392 m_stats.setNumGetSegments (m_stats.getNumGetSegments () + 1);
393 m_stats.setTotalGetSegmentSize (m_stats.getTotalGetSegmentSize () + segmentSize);
394
395 std::size_t numSegments = m_stats.getCurrentSegments () + 1;
396 std::size_t segmentSizeTotal = m_stats.getCurrentSegmentSize () + segmentSize;
397 m_stats.setCurrentSegments (numSegments);
398 m_stats.setCurrentSegmentSize (segmentSizeTotal);
399 if (m_stats.getPeakSegments () < numSegments)
400 {
401 m_stats.setPeakSegments (numSegments);
402 }
403 if (m_stats.getPeakSegmentSize () < segmentSizeTotal)
404 {
405 m_stats.setPeakSegmentSize (segmentSizeTotal);
406 }
407 }
408
410 void
411 logFreeSegment (void* segment, std::size_t segmentSize)
412 {
413 if (segment == nullptr)
414 {
415 return;
416 }
417 m_stats.setNumFreeSegments (m_stats.getNumFreeSegments () + 1);
418 m_stats.setTotalFreeSegmentSize (m_stats.getTotalFreeSegmentSize () + segmentSize);
419 m_stats.setCurrentSegments (m_stats.getCurrentSegments () - 1);
420 m_stats.setCurrentSegmentSize (m_stats.getCurrentSegmentSize () - segmentSize);
421 }
422
424 void
425 logAllocation (void* userPtr, std::size_t userSize, MemAllocType_et allocType)
426 {
427 if (userPtr != nullptr)
428 {
429 m_stats.setNumAllocations (m_stats.getNumAllocations () + 1);
430 m_stats.setTotalAllocated (m_stats.getTotalAllocated () + userSize);
431 std::size_t currentAllocated = m_stats.getCurrentAllocated () + userSize;
432 m_stats.setCurrentAllocated (currentAllocated);
433 if (m_stats.getPeakAllocated () < currentAllocated)
434 {
435 m_stats.setPeakAllocated (currentAllocated);
436 }
437 m_stats.setNumAllocations (allocType,
438 m_stats.getNumAllocations (allocType) + 1);
439 }
440 }
441
443 void
444 logReallocation (void* userPtr, std::size_t oldUserSize, std::size_t newUserSize)
445 {
446 if (userPtr != nullptr && newUserSize > oldUserSize)
447 {
448 std::size_t delta = newUserSize - oldUserSize;
449 m_stats.setTotalAllocated (m_stats.getTotalAllocated () + delta);
450 std::size_t currentAllocated = m_stats.getCurrentAllocated () + delta;
451 m_stats.setCurrentAllocated (currentAllocated);
452 if (m_stats.getPeakAllocated () < currentAllocated)
453 {
454 m_stats.setPeakAllocated (currentAllocated);
455 }
456 }
457 else if (userPtr != nullptr)
458 {
459 std::size_t delta = oldUserSize - newUserSize;
460 m_stats.setTotalDeallocated (m_stats.getTotalDeallocated () + delta);
461 m_stats.setCurrentAllocated (m_stats.getCurrentAllocated () - delta);
462 }
463 }
464
466 void
467 logDeallocation (void* userPtr, std::size_t userSize)
468 {
469 if (userPtr != nullptr)
470 {
471 m_stats.setNumDeallocations (m_stats.getNumDeallocations () + 1);
472 m_stats.setTotalDeallocated (m_stats.getTotalDeallocated () + userSize);
473 m_stats.setCurrentAllocated (m_stats.getCurrentAllocated () - userSize);
474 }
475 }
