300 for (std::size_t i = 0; i < MEM_ALLOC_COUNT; ++i)

302 std::printf (" %s: %zu\n",

357class StatsLogger
358{
359public:
361 void
362 logTrimEmptySegments ()
363 {
364 m_stats.setNumTrimEmptySegments (m_stats.getNumTrimEmptySegments () + 1);
365 }
366
368 void
369 logGetSegment (void* segment, std::size_t segmentSize)
370 {
371 if (segment == nullptr)
372 {
373 return;
374 }
375 m_stats.setNumGetSegments (m_stats.getNumGetSegments () + 1);
376 m_stats.setTotalGetSegmentSize (m_stats.getTotalGetSegmentSize () + segmentSize);
377
378 std::size_t numSegments = m_stats.getCurrentSegments () + 1;
379 std::size_t segmentSizeTotal = m_stats.getCurrentSegmentSize () + segmentSize;
380 m_stats.setCurrentSegments (numSegments);
381 m_stats.setCurrentSegmentSize (segmentSizeTotal);
382 if (m_stats.getPeakSegments () < numSegments)
383 {
384 m_stats.setPeakSegments (numSegments);
385 }
386 if (m_stats.getPeakSegmentSize () < segmentSizeTotal)
387 {
388 m_stats.setPeakSegmentSize (segmentSizeTotal);
389 }
390 }
391
393 void
394 logFreeSegment (void* segment, std::size_t segmentSize)
395 {
396 if (segment == nullptr)
397 {
398 return;
399 }
400 m_stats.setNumFreeSegments (m_stats.getNumFreeSegments () + 1);
401 m_stats.setTotalFreeSegmentSize (m_stats.getTotalFreeSegmentSize () + segmentSize);
402 m_stats.setCurrentSegments (m_stats.getCurrentSegments () - 1);
403 m_stats.setCurrentSegmentSize (m_stats.getCurrentSegmentSize () - segmentSize);
404 }
405
407 void
408 logAllocation (void* userPtr, std::size_t userSize, MemAllocType_et allocType)
409 {
410 if (userPtr != nullptr)
411 {
412 m_stats.setNumAllocations (m_stats.getNumAllocations () + 1);
413 m_stats.setTotalAllocated (m_stats.getTotalAllocated () + userSize);
414 std::size_t currentAllocated = m_stats.getCurrentAllocated () + userSize;
415 m_stats.setCurrentAllocated (currentAllocated);
416 if (m_stats.getPeakAllocated () < currentAllocated)
417 {
418 m_stats.setPeakAllocated (currentAllocated);
419 }
420 m_stats.setNumAllocations (allocType,
421 m_stats.getNumAllocations (allocType) + 1);
422 }
423 }
424
426 void
427 logReallocation (void* userPtr, std::size_t oldUserSize, std::size_t newUserSize)
428 {
429 if (userPtr != nullptr && newUserSize > oldUserSize)
430 {
431 std::size_t delta = newUserSize - oldUserSize;
432 m_stats.setTotalAllocated (m_stats.getTotalAllocated () + delta);
433 std::size_t currentAllocated = m_stats.getCurrentAllocated () + delta;
434 m_stats.setCurrentAllocated (currentAllocated);
435 if (m_stats.getPeakAllocated () < currentAllocated)
436 {
437 m_stats.setPeakAllocated (currentAllocated);
438 }
439 }
440 else if (userPtr != nullptr)
441 {
442 std::size_t delta = oldUserSize - newUserSize;
443 m_stats.setTotalDeallocated (m_stats.getTotalDeallocated () + delta);
444 m_stats.setCurrentAllocated (m_stats.getCurrentAllocated () - delta);
445 }
446 }
447
449 void
450 logDeallocation (void* userPtr, std::size_t userSize)
451 {
452 if (userPtr != nullptr)
453 {
454 m_stats.setNumDeallocations (m_stats.getNumDeallocations () + 1);
455 m_stats.setTotalDeallocated (m_stats.getTotalDeallocated () + userSize);
456 m_stats.setCurrentAllocated (m_stats.getCurrentAllocated () - userSize);
457 }
458 }
