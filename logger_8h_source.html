323class StatsLogger
324{
325public:
326 void
327 logCoalesce ()
328 {
329 m_stats.setNumCoalesces (m_stats.getNumCoalesces () + 1);
330 }
331
332 void
333 logGetSegment (void* segment, std::size_t segmentSize)
334 {
335 if (segment == nullptr)
336 {
337 return;
338 }
339 m_stats.setNumGetSegments (m_stats.getNumGetSegments () + 1);
340 m_stats.setTotalGetSegmentSize (m_stats.getTotalGetSegmentSize () + segmentSize);
341
342 std::size_t numSegments = m_stats.getCurrentSegments () + 1;
343 std::size_t segmentSizeTotal = m_stats.getCurrentSegmentSize () + segmentSize;
344 m_stats.setCurrentSegments (numSegments);
345 m_stats.setCurrentSegmentSize (segmentSizeTotal);
346 if (m_stats.getPeakSegments () < numSegments)
347 {
348 m_stats.setPeakSegments (numSegments);
349 }
350 if (m_stats.getPeakSegmentSize () < segmentSizeTotal)
351 {
352 m_stats.setPeakSegmentSize (segmentSizeTotal);
353 }
354 }
355
356 void
357 logFreeSegment (void* segment, std::size_t segmentSize)
358 {
359 if (segment == nullptr)
360 {
361 return;
362 }
363 m_stats.setNumFreeSegments (m_stats.getNumFreeSegments () + 1);
364 m_stats.setTotalFreeSegmentSize (m_stats.getTotalFreeSegmentSize () + segmentSize);
365 m_stats.setCurrentSegments (m_stats.getCurrentSegments () - 1);
366 m_stats.setCurrentSegmentSize (m_stats.getCurrentSegmentSize () - segmentSize);
367 }
368
369 void
370 logAllocation (void* userPtr, std::size_t userSize, MemAllocType_et allocType)
371 {
372 if (userPtr != nullptr)
373 {
374 m_stats.setNumAllocations (m_stats.getNumAllocations () + 1);
375 m_stats.setTotalAllocated (m_stats.getTotalAllocated () + userSize);
376 std::size_t currentAllocated = m_stats.getCurrentAllocated () + userSize;
377 m_stats.setCurrentAllocated (currentAllocated);
378 if (m_stats.getPeakAllocated () < currentAllocated)
379 {
380 m_stats.setPeakAllocated (currentAllocated);
381 }
382 m_stats.setNumAllocations (allocType,
383 m_stats.getNumAllocations (allocType) + 1);
384 }
385 }
386
387 void
388 logReallocation (void* userPtr, std::size_t oldUserSize, std::size_t newUserSize)
389 {
390 if (userPtr != nullptr && newUserSize > oldUserSize)
391 {
392 std::size_t delta = newUserSize - oldUserSize;
393 m_stats.setTotalAllocated (m_stats.getTotalAllocated () + delta);
394 std::size_t currentAllocated = m_stats.getCurrentAllocated () + delta;
395 m_stats.setCurrentAllocated (currentAllocated);
396 if (m_stats.getPeakAllocated () < currentAllocated)
397 {
398 m_stats.setPeakAllocated (currentAllocated);
399 }
400 }
401 else if (userPtr != nullptr)
402 {
403 std::size_t delta = oldUserSize - newUserSize;
404 m_stats.setTotalDeallocated (m_stats.getTotalDeallocated () + delta);
405 m_stats.setCurrentAllocated (m_stats.getCurrentAllocated () - delta);
406 }
407 }
408
409 void
410 logDeallocation (void* userPtr, std::size_t userSize)
411 {
412 if (userPtr != nullptr)
413 {
414 m_stats.setNumDeallocations (m_stats.getNumDeallocations () + 1);
415 m_stats.setTotalDeallocated (m_stats.getTotalDeallocated () + userSize);
416 m_stats.setCurrentAllocated (m_stats.getCurrentAllocated () - userSize);
417 }
418 }
419
420 void
421 logError (void*,
422 MemError_et error,
423 const char* file = __FILE__,
424 int line = __LINE__)
425 {
426 throw Exception (error, "memory corruption detected.", file, line);
427 }
428
429 MemStatistics& getStats () { return m_stats; }
430 const MemStatistics& getStats () const { return m_stats; }
431
432private:
433 MemStatistics m_stats;
434};
