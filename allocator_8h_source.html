407 pointer
408 allocate (size_type size, const void* = nullptr)
409 {
410 if (size > (std::numeric_limits<size_type>::max) () / sizeof (T))
411 {
412 throw std::bad_alloc ();
413 }
414 if (size == 0)
415 {
416 return nullptr;
417 }
418
419 // Decided here rather than in the class body, so that the allocator
420 // can still be named for an incomplete type T.
421 typedef std::integral_constant<bool, (alignof (T) > ALLOCATION_ALIGNMENT)> OverAligned;
422 void* ptr = allocateBytes (size * sizeof (T), OverAligned ());
423 if (ptr == nullptr)
424 {
425 throw std::bad_alloc ();
