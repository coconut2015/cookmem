217class Allocator

219public:

221 typedef typename ContextProvider::context_type context_type;
222 typedef T value_type;

224 typedef const T* const_pointer;

269
270 pointer
271 allocate (size_type size, const void* = nullptr)
272 {
273 if (size > (std::numeric_limits<size_type>::max) () / sizeof (T))
274 {
275 throw std::bad_alloc ();
276 }
277 if (size == 0)
278 {
279 return nullptr;
280 }
281
282 void* ptr = m_provider.get ().allocate (size * sizeof (T));
283 if (ptr == nullptr)
284 {
285 throw std::bad_alloc ();
286 }
287 return static_cast<pointer> (ptr);
288 }
289
290 void
291 deallocate (pointer ptr, size_type)
292 {
293 m_provider.get ().deallocate (ptr);
294 }
295
296 size_type
297 max_size () const
298 {
299 return (std::numeric_limits<size_type>::max) () / sizeof (T);
300 }
301

303 void
304 construct (U* ptr, Args&&... args)
305 {
306 ::new (static_cast<void*> (ptr)) U (std::forward<Args> (args)...);
307 }
308
309 template<class U>
310 void
311 destroy (U* ptr)
312 {
313 ptr->~U ();
314 }

317 getContext () const

319 return m_provider.get ();
320 }
