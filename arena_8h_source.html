68class FixedArena
69{
70public:
79 FixedArena (void* ptr, std::size_t size)
80 : m_used (false)
81 {
82 std::size_t remain = (std::size_t)ptr & 0xfUL;
83 if (remain)
84 {
85 std::size_t inc = (0x10 - remain);
86 m_page = (void*)((char*)ptr + inc);
87 size -= inc;
88 }
89 else
90 {
91 m_page = ptr;
92 }
93 size &= 0xfffffffffffffff0UL;
94 m_size = size;
95 }

105 void*
106 getSegment (std::size_t& size)
107 {
108 if (m_used || size > m_size)
109 {
110 return NULL;
111 }
112 m_used = true;
113 size = m_size;
114 return m_page;
115 }
116
126 bool
127 freeSegment (void* ptr, std::size_t size)
128 {
129 if (!m_used || ptr != m_page || size != m_size)
130 {
131 return true;
132 }
133 m_used = false;
134 return false;
135 }

137private:
138 void* m_page;
139 std::size_t m_size;
