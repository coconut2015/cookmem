78class FixedArena
79{
80public:
93 FixedArena (void* ptr, std::size_t size)
94 : m_used (false)
95 {
96 const std::size_t mask = ALLOCATION_ALIGNMENT - 1;
97 std::size_t remain = reinterpret_cast<std::uintptr_t> (ptr) & mask;
98 if (remain)
99 {
100 std::size_t inc = ALLOCATION_ALIGNMENT - remain;
101 m_page = static_cast<char*> (ptr) + inc;
102 // A buffer smaller than the bytes skipped has no usable space.
103 size = (size > inc) ? size - inc : 0;
104 }
105 else
106 {
107 m_page = ptr;
108 }
109 size &= ~mask;
110 m_size = size;
111 }
112
121 void*
122 getSegment (std::size_t& size)
123 {
124 if (m_used || size > m_size)
125 {
126 return NULL;
127 }
128 m_used = true;
129 size = m_size;
130 return m_page;
131 }
132
142 bool
143 freeSegment (void* ptr, std::size_t size)
144 {
145 if (!m_used || ptr != m_page || size != m_size)
146 {
147 return true;
148 }
149 m_used = false;
150 return false;
151 }
152
157 void
158 releaseAll ()
159 {
160 }
161
162private:
163 void* m_page;
164 std::size_t m_size;
165 bool m_used;
166};
