64class FixedArena
65{
66public:
75 FixedArena (void* ptr, std::size_t size)
76 : m_used (false)
77 {
78 std::size_t remain = (std::size_t)ptr & 0xfUL;
79 if (remain)
80 {
81 std::size_t inc = (0x10 - remain);
82 m_page = (void*)((char*)ptr + inc);
83 size -= inc;
84 }
85 else
86 {
87 m_page = ptr;
88 }
89 size &= 0xfffffffffffffff0UL;
90 m_size = size;
91 }
92
101 void*
102 getSegment (std::size_t& size)
103 {
104 if (m_used || size > m_size)
105 {
106 return NULL;
107 }
108 m_used = true;
109 size = m_size;
110 return m_page;
111 }
112
122 bool
123 freeSegment (void* ptr, std::size_t size)
124 {
125 if (!m_used || ptr != m_page || size != m_size)
126 {
127 return true;
128 }
129 m_used = false;
130 return false;
131 }
132
133private:
134 void* m_page;
135 std::size_t m_size;
136 bool m_used;
137};
