172class StatsLogger
173{
174public:
175 StatsLogger ()
176 : m_numAllocations (0),
177 m_numDeallocations (0),
178 m_numGetSegments (0),
179 m_numFreeSegments (0),
180 m_totalAllocated (0),
181 m_totalDeallocated (0),

183 m_totalFreeSegmentSize (0),
184 m_numAllocationsByType {}
185 {
186 }
187
188 void
189 logGetSegment (void*, std::size_t segmentSize)
190 {
191 ++m_numGetSegments;
192 m_totalGetSegmentSize += segmentSize;
193 }
194
195 void
196 logFreeSegment (void*, std::size_t segmentSize)
197 {
198 ++m_numFreeSegments;
199 m_totalFreeSegmentSize += segmentSize;
200 }
201
202 void
203 logAllocation (void* userPtr, std::size_t userSize, MemAllocType_et allocType)
204 {
205 if (userPtr != nullptr)
206 {
207 ++m_numAllocations;
208 m_totalAllocated += userSize;
209 ++m_numAllocationsByType[allocType];
210 }
211 }
212
213 void
214 logReallocation (void*, std::size_t oldUserSize, std::size_t newUserSize)
215 {
216 if (newUserSize > oldUserSize)
217 {
218 m_totalAllocated += newUserSize - oldUserSize;
219 }
220 else
221 {
222 m_totalDeallocated += oldUserSize - newUserSize;
223 }
224 }
225
226 void
227 logDeallocation (void* userPtr, std::size_t userSize)
228 {
229 if (userPtr != nullptr)
230 {
231 ++m_numDeallocations;
232 m_totalDeallocated += userSize;
233 }
234 }
235
236 void
237 logError (void*,
238 MemError_et error,
239 const char* file = __FILE__,
240 int line = __LINE__)
241 {
242 throw Exception (error, "memory corruption detected.", file, line);
243 }
