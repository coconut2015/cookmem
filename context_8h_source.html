204 template<class U, class... Args>
205 U*
206 create (Args&&... args)
207 {
208 void* mem = m_pool.allocateAligned (sizeof (U), alignof (U));
209 if (mem == nullptr)
210 {
211 throw std::bad_alloc ();
212 }
213 try
214 {
215 return ::new (mem) U (std::forward<Args> (args)...);
216 }
217 catch (...)
218 {
219 m_pool.deallocate (static_cast<T*> (mem));
220 throw;
221 }
222 }

234 template<class U>
235 void
236 destroy (U* ptr)
237 {
238 if (ptr == nullptr)
239 {
240 return;
241 }
242 // Find the start of the allocation before the destructor runs: with
243 // multiple inheritance, a base class pointer can point inside it.
244 void* mem = getObjectStart (ptr, std::is_polymorphic<U> ());
245 ptr->~U ();
246 m_pool.deallocate (static_cast<T*> (mem));
247 }
